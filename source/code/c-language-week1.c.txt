#include <stdio.h>

static int binary_search(const int arr[], int size, int target)
{
    int left = 0;
    int right = size - 1;
    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (arr[mid] < target) left = mid + 1;
        else if (arr[mid] > target) right = mid - 1;
        else return mid;
    }
    return -1;
}

int main(void)
{
    const int numbers[] = {1, 3, 5, 7, 9, 11, 13};
    const int size = (int)(sizeof(numbers) / sizeof(numbers[0]));
    const int index = binary_search(numbers, size, 7);
    printf("%s\n", index >= 0 ? "找到目标" : "没有找到目标");
    return 0;
}
